      runSingle(suiteId) {
        const suite = this.suites.find(s => s.id === suiteId);
        if (!suite) return;
        const mock = this.createIsolatedMockBackend();
        const t0 = performance.now();
        try {
          const detail = suite.run(mock);
          const ms = (performance.now() - t0).toFixed(1);
          this.results[suite.id] = { passed: true, detail, ms };
        } catch (err) {
          const ms = (performance.now() - t0).toFixed(1);
          this.results[suite.id] = { passed: false, detail: err.message, ms };
        }
        this.render();
      },

      runAll() {
        this.suites.forEach(s => this.runSingle(s.id));
        const passedCount = Object.values(this.results).filter(r => r.passed).length;
        UIRenderer.showToast(`🧪 Suíte concluída: ${passedCount}/${this.suites.length} testes aprovados!`);
      },

      render() {
        const listEl = document.getElementById('integration-tests-list');
        if (!listEl) return;

        const values = Object.values(this.results);
        const passed = values.filter(r => r.passed).length;
        const failed = values.filter(r => !r.passed).length;

        document.getElementById('test-kpi-total').textContent = this.suites.length;
        document.getElementById('test-kpi-passed').textContent = passed;
        document.getElementById('test-kpi-failed').textContent = failed;

        listEl.innerHTML = this.suites.map(s => {
          const res = this.results[s.id];
          let badge = `Aguardando`;
          let borderCls = 'border-zinc-800';
          if (res) {
            if (res.passed) {
              badge = `PASS • ${res.ms}ms`;
              borderCls = 'border-emerald-500/40';
            } else {
              badge = `FAIL • ${res.ms}ms`;
              borderCls = 'border-rose-500/40';
            }
          }

          return `
            
          
              
          
                
          
                  
          ${UIRenderer.escapeHtml(s.title)}
          
                  ${badge}
                
          
                
          ${UIRenderer.escapeHtml(s.desc)}
          
                ${res ? `
                  
          
                    ${res.passed ? '✓ ' : '✗ '}${UIRenderer.escapeHtml(res.detail)}
                  
          
                ` : ''}
              
          
              
            
          
          `;
        }).join('');
      }
